# 全文检索数据存储系统 - 依赖
# 核心逻辑仅依赖 Python 标准库，以下为可选格式解析依赖

# .docx 解析（不装则无法导入 docx）
python-docx>=1.1.0

# .pdf 解析（不装则无法导入 pdf）
pypdf>=4.0.0

# ===== 语义检索通道（向量召回）=====
# 安装后，导入完成后会自动在后台构建向量索引；
# 不安装时系统会自动跳过语义通道，仅使用关键词/关联词检索，不影响主流程。
#
# 模型说明：
#   - 默认使用 bge-small-zh（约 95MB，384 维向量，中文优化）
#   - 首次运行时会从 HuggingFace 自动下载模型到本地缓存（约 95MB）
#   - 之后完全离线运行，无需联网
#
# 资源占用（参考值，10.5 万 chunk / 4200 万字）：
#   - 索引构建：约 10~15 分钟（Xeon Silver 4114）
#   - 内存占用：< 1 GB
#   - 单次查询：< 50 ms
#   - 磁盘占用：约 150 MB（向量 + HNSW 索引文件）

# 向量模型加载与推理（bge-small-zh 等 Sentence Transformer 模型）
sentence-transformers>=2.5.0

# 向量索引引擎（HNSW 算法，CPU 版本，无需 Docker）
faiss-cpu>=1.7.4

# 数值计算（sentence-transformers 与 faiss 的依赖，通常会被自动安装）
numpy>=1.24.0
